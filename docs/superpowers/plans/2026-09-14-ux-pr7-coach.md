### Task 1: `lib/timeBudget.ts`

```ts
export interface BudgetSegment { cls: CoachIntervalClass | "unclassified"; seconds: number; count: number; share: number; avg: number | null; vsMatch: number | null; outliers: { shotNumber: number; seconds: number }[] }
export interface TimeBudget { total: number; segments: BudgetSegment[] /* taxonomy order, zero-count omitted */; outlierCount: number; classified: boolean }
export function timeBudget(shots: CoachShot[], distributions: CoachMatchDistributions | null): TimeBudget
export function matchBudget(perStage: { stageNumber; stageName; shots; audited }[], distributions): { stages: (TimeBudget & { stageNumber; stageName })[]; maxTotal; shareByClass: Record<cls, number>; outlierCount }
export const BUDGET_ORDER: CoachIntervalClass[]   // first_shot, movement, transition, split, reload, activation (verify against the union after the merge)
export const BUDGET_TICK: Record<cls | "unclassified", ChipTick | "none">
```
Rules: each shot's `split` accrues to its `interval_class` (the draw is `first_shot`; `null` class -> `unclassified`); `total = sum(split)` and must equal the last shot's `time_from_beep` within 1 ms (assert in tests with the stage-03 fixture from `scripts/seed_demo_match.py` STAGE3_T / STAGE3_CLS); `share = seconds / total`; `avg = seconds / count`; `vsMatch = avg - distributions[cls].mean_s` when both exist; outlier when `split > 2 * distributions[cls].median_s` (null median -> none); `classified` when any shot has a class. `matchBudget` scales bars by `maxTotal`.
